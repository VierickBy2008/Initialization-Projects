#include <iostream>
#include <string>

using namespace std;

int main() {
    int numAlumnos;
    float PC, Parcial_1, PC2, Parcial_Final, EC1, EC2, Nota_final, EP1, EP2;
    string nombre;

    cout << "--- SISTEMA DE GESTION ACADEMICA (VERSION 2.0) ---" << endl;
    cout << "Ingrese la cantidad de alumnos a evaluar: ";
    cin >> numAlumnos;

    for (int i = 1; i <= numAlumnos; i++) {
        cout << "\n===================================" << endl;
        cout << "DATOS DEL ALUMNO #" << i << endl;
        cout << "Nombre del estudiante: ";
        cin.ignore(); 
        getline(cin, nombre);

        // --- Validacion Periodo 1 ---
        cout << "\nNotas de " << nombre << " - Periodo 1:" << endl;
        do {
            cout << "EC1 (0-20): "; cin >> EC1;
        } while (EC1 < 0 || EC1 > 20);

        do {
            cout << "PC1 (0-20): "; cin >> PC;
        } while (PC < 0 || PC > 20);

        do {
            cout << "Parcial 1 (0-20): "; cin >> Parcial_1;
        } while (Parcial_1 < 0 || Parcial_1 > 20);

        // Calculo directo de EP1
        EP1 = (EC1 * 0.15) + (PC * 0.40) + (Parcial_1 * 0.45);

        // --- Validacion Periodo 2 ---
        cout << "\nNotas de " << nombre << " - Periodo 2:" << endl;
        do {
            cout << "EC2 (0-20): "; cin >> EC2;
        } while (EC2 < 0 || EC2 > 20);

        do {
            cout << "PC2 (0-20): "; cin >> PC2;
        } while (PC2 < 0 || PC2 > 20);

        do {
            cout << "Final (0-20): "; cin >> Parcial_Final;
        } while (Parcial_Final < 0 || Parcial_Final > 20);

        // Calculo directo de EP2
        EP2 = (EC2 * 0.15) + (PC2 * 0.40) + (Parcial_Final * 0.45);

        // --- Resultado Final ---
        Nota_final = (EP1 + EP2) / 2;

        cout << "\n>>> RESULTADO FINAL DE " << nombre << " <<<" << endl;
        cout << "Promedio: " << Nota_final << endl;

        if (Nota_final > 10.5) {
            cout << "ESTADO: APROBADO. ¡Felicidades!" << endl;
        }
        else if (Nota_final >= 9.5 && Nota_final <= 10.5) {
            cout << "ESTADO: SUSTITUTORIO. Apto para el examen." << endl;
        }
        else {
            cout << "ESTADO: DESAPROBADO. Mas suerte para la proxima." << endl;
        }
    }

    cout << "\n===================================" << endl;
    cout << "Proceso completado. ¡Exitos en tus estudios!" << endl;

    return 0;
}
