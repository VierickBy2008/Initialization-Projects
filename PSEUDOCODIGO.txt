Algoritmo CalcularPromedioFinal
	
	Definir ec1, pc1, par1, ep1 Como Real
	Definir ec2, pc2, par2, ep2 Como Real
	Definir notaFinal Como Real
	
	// --- Evaluación Periodo 1 ---
	Escribir "Ingrese su nota de Evaluación continua 1: "
	Leer ec1
	Escribir "Ingrese su nota de la PC 1: "
	Leer pc1
	Escribir "Ingrese su nota del Parcial 1: "
	Leer par1
	
	ep1 <- (ec1 * 0.15) + (pc1 * 0.40) + (par1 * 0.45)
	
	// --- Evaluación Periodo 2 ---
	Escribir "Ingrese su nota de Evaluación continua 2: "
	Leer ec2
	Escribir "Ingrese su nota de la PC 2: "
	Leer pc2
	Escribir "Ingrese su nota del Parcial Final: "
	Leer par2
	
	ep2 <- (ec2 * 0.15) + (pc2 * 0.40) + (par2 * 0.45)
	
	// --- Cálculo Final ---
	notaFinal <- (ep1 + ep2) / 2
	
	Escribir "Tu nota final del curso es igual a: ", notaFinal

		Sino
